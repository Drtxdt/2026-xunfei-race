# 致所有人：登录小车后，先
cd 2026-xunfei-race

# 可选：构建（必须强制使用系统 Python 3.7，否则 cv_bridge 等 Python 绑定会链接到 conda
# 的 Python 3.9 并导致 strict_mission 等节点启动时 SIGABRT 崩溃）
catkin_make \
  -DPYTHON_EXECUTABLE=/usr/bin/python3 \
  -DPYTHON_INCLUDE_DIR=/usr/include/python3.7m \
  -DPYTHON_LIBRARY=/usr/lib/aarch64-linux-gnu/libpython3.7m.so \
  -DPYTHON_NUMPY_INCLUDE_DIR=/usr/lib/python3/dist-packages/numpy/core/include
# 指定某一个包构建
catkin_make --pkg ucar_2026_line_follow

# 必做：source环境
source devel/setup.bash

# 左侧巡线
roslaunch ucar_2026_line_follow line_follow_test.launch

roslaunch ucar_2026_line_follow straight_right.launch

roslaunch ucar_2026_line_follow right_line_follow_cpp.launch
# c++中间巡线
roslaunch ucar_2026_line_follow straight_right_cpp.launch
